\begin{frame}{Problem Setup}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \large
      Two observed systems contain their own dynamics plus a shared imprint from an unobserved driver.
      \vspace{0.8em}
      \begin{itemize}
        \item Observed time series: \textbf{x} and \textbf{y}.
        \item Hidden process: \textbf{z}, a common driver.
        \item Time-delay embedding reconstructs state spaces from histories.
        \item The target is the latent trajectory, up to an admissible transformation.
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{block}{Fork structure}
        \centering
        \Huge\textcolor{DeckMagenta}{$z$}\\[-0.2em]
        \Large\textcolor{DeckMuted}{hidden driver}\\[1.2em]
        \Huge
        \textcolor{DeckTeal}{$\swarrow$}\hspace{1.8em}\textcolor{DeckTeal}{$\searrow$}\\[0.2em]
        \textcolor{DeckOrange}{$x$}\hspace{2.2em}\textcolor{DeckTeal}{$y$}\\[-0.2em]
        \normalsize\textcolor{DeckMuted}{observed systems}
      \end{block}
    \end{column}
  \end{columns}
  \source{Both cited papers exploit the redundant imprint of a shared latent driver in driven observations.}
\end{frame}

\begin{frame}{Method Landscape}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \textbf{MaCo}: Mapper-Coach neural bottleneck trained through prediction.
        \item \textbf{ASOM/AniSOM}: anisotropic self-organizing map for shared/self dynamics.
        \item \textbf{ShRec}: recurrence-structure shared reconstruction baseline.
        \item \textbf{DCCA}: deep canonical correlation analysis.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \textbf{DCA}: dynamical component analysis.
        \item \textbf{CCA}: linear correlated-components baseline.
        \item \textbf{PCA / ICA / kPCA}: classical extraction and source separation.
        \item \textbf{SFA / Random}: slow-feature method and sanity-check control.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{block}{Comparison question}
    Which kind of structure best exposes the hidden driver: predictive bottlenecks, attractor geometry, correlated features, or generic variance/source decomposition?
  \end{block}
\end{frame}

\begin{frame}{Mapper-Coach Intuition}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.38\textwidth}
      MaCo learns to predict one driven system using its own past and information extracted from the other driven system.
      \vspace{0.8em}
      \begin{itemize}
        \item Mapper compresses the reconstructed state of one observation.
        \item Coach uses that bottleneck to predict the other observation.
        \item Successful prediction makes the bottleneck align with the hidden driver.
      \end{itemize}
    \end{column}
    \begin{column}{0.58\textwidth}
      \centering
      \includegraphics[width=\textwidth,height=0.60\textheight,keepaspectratio]{example_logmap_res.png}
    \end{column}
  \end{columns}
  \source{Example logistic-map run: high prediction quality and high latent reconstruction quality appear together.}
\end{frame}

\begin{frame}{ASOM / AniSOM Intuition}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \large
      The ASOM paper frames hidden-driver inference through time-delay embedding, intrinsic dimension, mutual dimension, and anisotropic SOM training.
      \vspace{0.8em}
      \begin{itemize}
        \item The observed attractor is treated as a structured manifold.
        \item Anisotropic training emphasizes directions associated with common dynamics.
        \item The output gives a geometry-first unsupervised driver estimate.
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{block}{Conceptual split}
        \small
        Observed manifold: \metric{self + shared}\\
        Mutual structure: \metric{shared driver}\\
        AniSOM output: \metric{driver estimate}
      \end{block}
      \vspace{0.5em}
      \small\color{DeckMuted}
      This complements MaCo: one method learns through prediction, the other through attractor geometry.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Interpretation}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.32\textwidth}
      \begin{block}{Prediction can reveal latent structure}
        MaCo supports the Mapper-Coach claim: a minimal predictive bottleneck can align with the common hidden driver.
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Geometry remains competitive}
        ASOM/AniSOM offers a strong unsupervised manifold baseline, especially on discrete chaotic maps.
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Continuous flows are harder}
        Lorenz results show that method ranking changes when dynamics and embeddings become more complex.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \large
  Together, the papers motivate both neural and geometric approaches to hidden-driver inference.
\end{frame}

\begin{frame}{Takeaways and References}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.43\textwidth}
      \begin{itemize}
        \item Hidden common drivers are recoverable when their imprint is redundantly encoded.
        \item MaCo is strongest on the discrete-map benchmarks in this repo.
        \item Lorenz and noise analyses show where reconstruction becomes more demanding.
      \end{itemize}
